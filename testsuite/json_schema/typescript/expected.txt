procedure Free is new Ada.Unchecked_Deallocation
(User_Array, User_Array_Access);

overriding procedure Adjust (Self : in out User_Vector) is
begin
if Self.Length > 0 then
Self.Data :=
new User_Array'(Self.Data (1 .. Self.Length));
end if;
end Adjust;

overriding procedure Finalize (Self : in out User_Vector) is
begin
Free (Self.Data);
Self.Length := 0;
end Finalize;

function Empty return User_Vector is (Ada.Finalization.Controlled with others => <>);

function Is_Null (Self : User_Vector) return Boolean is (Self.Is_Null);

function Length (Self : User_Vector) return Natural is (Self.Length);

procedure Clear (Self : in out User_Vector; Is_Null : Boolean := True) is
begin
   Self.Length := 0;
   Self.Is_Null := Is_Null;
end Clear;

procedure Append
(Self : in out User_Vector;
Value : User) is
Init_Length : constant Positive :=
Positive'Max (2, 256 / User'Size);
Self_Data_Saved : User_Array_Access := Self.Data;
begin
if Self.Length = 0 then
Self.Is_Null := False;
Self.Data := new User_Array (1 .. Init_Length);
elsif Self.Length = Self.Data'Last then
Self.Data :=
new User_Array (1 .. 3 * Self.Length / 2 + 1);
Self.Data (1 .. Self.Length) := Self_Data_Saved.all;
Free (Self_Data_Saved);
end if;
Self.Length := Self.Length + 1;
Self.Data (Self.Length) := Value;
end Append;

pragma Style_Checks (Off);
procedure Output_Any_Value
  (Handler : in out VSS.JSON.Content_Handlers.JSON_Content_Handler'Class;
   Value   : Any_Value'Class) is
begin
   for Item of Value loop
      case Item.Kind is
         when VSS.JSON.Streams.Start_Array =>
            Handler.Start_Array;
         when VSS.JSON.Streams.End_Array =>
            Handler.End_Array;
         when VSS.JSON.Streams.Start_Object =>
            Handler.Start_Object;
         when VSS.JSON.Streams.End_Object =>
            Handler.End_Object;
         when VSS.JSON.Streams.Key_Name =>
            Handler.Key_Name (Item.Key_Name);
         when VSS.JSON.Streams.String_Value =>
            Handler.String_Value (Item.String_Value);
         when VSS.JSON.Streams.Number_Value =>
            Handler.Number_Value (Item.Number_Value);
         when VSS.JSON.Streams.Boolean_Value =>
            Handler.Boolean_Value (Item.Boolean_Value);
         when VSS.JSON.Streams.Null_Value =>
            Handler.Null_Value;
         when VSS.JSON.Streams.None =>
            null;
         when VSS.JSON.Streams.Invalid =>
            raise Program_Error;
         when VSS.JSON.Streams.Start_Document =>
            raise Program_Error;
         when VSS.JSON.Streams.End_Document =>
            raise Program_Error;
         when VSS.JSON.Streams.Comment =>
            raise Program_Error;
      end case;
   end loop;
end Output_Any_Value;

procedure Output_Scope
(Handler : in out VSS.JSON.Content_Handlers.JSON_Content_Handler'Class;
Value : Scope) is
begin
case Value is
when messages_read =>
Handler.String_Value ("messages.read");
when guilds_join =>
Handler.String_Value ("guilds.join");
end case;
end Output_Scope;

procedure Output_Shape
(Handler : in out VSS.JSON.Content_Handlers.JSON_Content_Handler'Class;
Value : Shape) is
begin
case Value.Union.Kind is
when circle_Variant =>
Output_Circle (Handler, Value.Union.circle_Variant);
when square_Variant =>
Output_Square (Handler, Value.Union.square_Variant);
end case;
end Output_Shape;

procedure Output_Circle
(Handler : in out VSS.JSON.Content_Handlers.JSON_Content_Handler'Class;
Value : Circle) is
begin
Handler.Start_Object;
Handler.Key_Name ("shape");Handler.String_Value ("circle");
Handler.Key_Name ("radius");
Handler.Integer_Value(Value.radius);
Handler.End_Object;
end Output_Circle;

procedure Output_Square
(Handler : in out VSS.JSON.Content_Handlers.JSON_Content_Handler'Class;
Value : Square) is
begin

Handler.Key_Name ("shape");Handler.String_Value ("square");

Handler.Integer_Value(Value.side);
Handler.End_Object;
end Output_Square;

procedure Output_User
(Handler : in out VSS.JSON.Content_Handlers.JSON_Content_Handler'Class;
Value : User) is
begin
Handler.Start_Object;
Handler.Key_Name ("id");
Handler.String_Value (Value.id);
Handler.Key_Name ("avatar");
if Value.avatar.Is_Null then
Handler.Null_Value;
else
Handler.String_Value (Value.avatar);
end if;
if Value.level.Is_Set then
Handler.Key_Name ("level");
Handler.Integer_Value(Value.level.Value);
end if;
Handler.End_Object;
end Output_User;

procedure Output_Message
(Handler : in out VSS.JSON.Content_Handlers.JSON_Content_Handler'Class;
Value : Message) is
procedure Output_Message_application
(Handler : in out VSS.JSON.Content_Handlers.JSON_Content_Handler'Class;
Value : Message_application) is
begin
Handler.Start_Object;
if not Value.id.Is_Null then
Handler.Key_Name ("id");
Handler.String_Value (Value.id);
end if;
Handler.End_Object;
end Output_Message_application;

begin
Handler.Start_Object;
Handler.Key_Name ("id");
Handler.String_Value (Value.id);
if Value.author.Is_Set then
Handler.Key_Name ("author");
Output_User (Handler, Value.author.Value);
end if;
Handler.Key_Name ("channel");
Output_Channel (Handler, Value.channel);
Handler.Key_Name ("shape");
Output_Shape (Handler, Value.shape);
if Value.layout.Is_Set then
Handler.Key_Name ("layout");
Handler.Integer_Value (1);
end if;
if Value.scope.Is_Set then
Handler.Key_Name ("scope");
Output_Scope (Handler, Value.scope.Value);
end if;
if Value.count.Is_Set then
Handler.Key_Name ("count");
Handler.Integer_Value(Value.count.Value);
end if;
if not Value.nonce.Is_Empty then
Handler.Key_Name ("nonce");
Output_Any_Value (Handler, Value.nonce);
end if;
if not Value.mentions.Is_Empty then
Handler.Key_Name ("mentions");
Handler.Start_Array;
for J in 1 .. Value.mentions.Length loop
Handler.String_Value (Value.mentions (J));
end loop;
Handler.End_Array;
end if;
if not Value.levels.Is_Null then
Handler.Key_Name ("levels");
Handler.Start_Array;
for J in 1 .. Value.levels.Length loop
Handler.Integer_Value(Value.levels (J));
end loop;
Handler.End_Array;
end if;
if not Value.extras.Is_Empty then
Handler.Key_Name ("extras");
Output_Any_Value (Handler, Value.extras);
end if;
if not Value.Item_0.Is_Null then
Handler.Key_Name ("0");
Handler.String_Value (Value.Item_0);
end if;
if Value.application.Is_Set then
Handler.Key_Name ("application");
Output_Message_application (Handler, Value.application.Value);
end if;
if Value.referenced_message.Length > 0 then
Handler.Key_Name ("referenced_message");
Output_Message (Handler, Value.referenced_message.Value);
end if;
Handler.End_Object;
end Output_Message;

Output_Any_Value (Handler, Value);
end Output_Mixed;

procedure Output_DMChannel

if not Value.recipients.Is_Null then
Handler.Key_Name ("recipients");

for J in 1 .. Value.recipients.Length loop
Output_User (Handler, Value.recipients (J));
end loop;
Handler.End_Array;
end if;
Handler.End_Object;
end Output_DMChannel;

when DMChannel_Variant =>
Output_DMChannel (Handler, Value.Union.DMChannel_Variant);
end case;
end Output_Channel;

(Handler : in out VSS.JSON.Content_Handlers.JSON_Content_Handler'Class;

Handler.String_Value (Value.topic);
end if;
Handler.End_Object;
end Output_BaseChannel_0;

procedure Input_Any_Value
  (Reader  : in out VSS.JSON.Pull_Readers.JSON_Pull_Reader'Class;
   Value   : out Any_Value'Class;
   Success : in out Boolean) is
use type VSS.JSON.Streams.JSON_Stream_Element_Kind;
begin
case Reader.Element_Kind is
when VSS.JSON.Streams.Start_Array =>
Value.Append ((Kind => VSS.JSON.Streams.Start_Array));
Reader.Read_Next;
while Success and Reader.Element_Kind /= VSS.JSON.Streams.End_Array loop
Input_Any_Value (Reader, Value, Success);
end loop;
